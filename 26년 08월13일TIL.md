    #include <iostream>
    #include <vector>

    using namespace std;

    int Calculator(const vector<int>& weights, int totalWeight, int leftWeights, bool checkindex[], int rightWeight = 0, int leftWeight = 0)
    {

        const int factorial[10] = { 1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880 };
        int leftDatas = totalWeight - rightWeight - leftWeight;
        int answer = 0;


        if (leftWeights == 0) { return 0; }


        if (rightWeight + leftDatas <= leftWeight) { return 0; }


        for (int loop = 0; loop < weights.size(); loop = loop + 1)
        {
            if (weights.size() == leftWeights)
            {
                checkindex[loop] = true;
                answer = answer + Calculator(weights, totalWeight, leftWeights - 1, checkindex, rightWeight, leftWeight + weights[loop]);
                checkindex[loop] = false;
                continue;
            }
            if (checkindex[loop] == true)
            {
                continue;
            }
            else
            {
                checkindex[loop] = true;
                answer = answer + Calculator(weights, totalWeight, leftWeights - 1, checkindex, rightWeight, leftWeight + weights[loop]);

                if (rightWeight + weights[loop] > leftWeight)
                {
                    answer = answer + (1 <<  (leftWeights - 1)) * factorial[leftWeights - 1];
                }
                else
                {
                    answer = answer + Calculator(weights, totalWeight, leftWeights - 1, checkindex, rightWeight + weights[loop], leftWeight);
                }
                checkindex[loop] = false;
            }
        }
        return answer;
    }

    int main(int argc, char** argv)
    {

        const int factorial[10] = { 0, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880 };
        int countWeight = 0;
        int totalWeight = 0;

        int T = 0;
        cin >> T;


        for (int loop = 0; loop < T; loop = loop + 1)
        {

            cin >> countWeight;

            vector<int> weights;
            int answer = (1 << (countWeight - 1)) * factorial[countWeight]; // 처음부터 오른 쪽에 올리는 경우의 수 제외 기본 식은 (2 ^ (N - 1)) * N!

            for (int index = 0; index < countWeight; index = index + 1)
            {
                int buf = 0;
                cin >> buf;
                weights.push_back(buf);
                totalWeight = totalWeight + buf;
            }

            bool checkindex[9] = { false };
            int check = Calculator(weights, totalWeight, weights.size(), checkindex);
            answer = answer - check;

            totalWeight = 0;
            cout << "#" << loop + 1 << " " << answer << "\n";

        }

        return 0;//정상종료시 반드시 0을 리턴해야합니다.

    }
