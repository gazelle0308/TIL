    // 작성한 코드
    #include<iostream>
    #include<vector>
    #include<set>

    using namespace std;

    struct FRoadInfo
    {
        int point = 0;
        set<int> hour;
    };

    void solution (int vIndex, int start, int beforeHour, vector<vector<FRoadInfo>>& target, const vector<vector<FRoadInfo>>& serch, vector<int>& saveIndex, vector<bool> check = vector<bool>(1000, false))
    {

        if(check[vIndex])
        { return; }
        else
        { check[vIndex] = true; }

        for (int sLoop = 0; sLoop < serch[vIndex].size(); sLoop = sLoop + 1)
        {
            if (vIndex != start)
            { 

                bool isGet = false;
                bool pass = false;

                int rPoint = serch[vIndex][sLoop].point;

                FRoadInfo* uploadRoadInfo = new FRoadInfo;


                // 리버스 그래프 기준 내가 위치 = partyPlace랑 연결된 T 혹은 T와 연결된 Y인 상황(후속으로 Y는 T로 병합됨)
                // 리버스 그래프 기준 내가 확인해야 하는 것 = T혹은 Y가 partyPlace랑 이미 연결되었는가? = 첫번 째 인덱스는 Y혹은 T = 두번째 인덱스를 순환하면서 요소들의 구조체 FRoadInfo의 point가 T나 partyPlace인지 확인해야함
                for (int tLoop = 0; tLoop < target[rPoint].size(); tLoop = tLoop + 1)
                {

                    if (target[rPoint][tLoop].point == start)
                    {

                        if (*target[rPoint][tLoop].hour.begin() < beforeHour + *serch[vIndex][sLoop].hour.begin())
                        {
                            delete uploadRoadInfo;
                            pass = true;
                            break; 
                        }

                        delete uploadRoadInfo;

                        uploadRoadInfo = &target[rPoint][tLoop];
                        isGet = true;
                        break;
                    }

                }
                if (!pass)
                {

                    if (isGet)
                    {
                        uploadRoadInfo->hour.insert(beforeHour + *serch[vIndex][sLoop].hour.begin());
                    }
                    else
                    {
                        uploadRoadInfo->point = start;
                        uploadRoadInfo->hour.insert(beforeHour + *serch[vIndex][sLoop].hour.begin());
                        target[rPoint].push_back(*uploadRoadInfo);
                        saveIndex[rPoint] = target[rPoint].size() - 1;

                        delete uploadRoadInfo;

                    }

                }

            }

            solution(serch[vIndex][sLoop].point, start, beforeHour + *serch[vIndex][sLoop].hour.begin(), target, serch, saveIndex, check);

        }

    }

    int main(int argc, char** argv)
    {
        int test_case;

        // T
        int caseT;

        // N
        int numOfHouses;
        // M
        int numOfRoads;
        // X
        int partyPlace;

        int answer;

        // N의 최대값이 1000이므로, 최대 1000개의 집을 수용할 수 있도록 미리 1000개의 원소를 확보하려고 1000을 지정
        vector<vector<FRoadInfo>> graph(1000);
        vector<vector<FRoadInfo>> rGraph(1000);
        vector<int> writeIndex(1000, 0);
        vector<int> rWriteIndex(1000, 0);

        cin >> caseT;


        for (test_case = 1; test_case <= caseT; test_case = test_case + 1)
        {

            set<int, greater<int>> findFastPath;

            cin >> numOfHouses >> numOfRoads >> partyPlace;

            for (int inputLoop = 0; inputLoop < numOfRoads; inputLoop = inputLoop + 1)
            {

                // 인덱스 = 집 번호
                int startPoint = 0;
                int endPoint = 0;
                int hour = 0;

                FRoadInfo info;
                FRoadInfo rInfo;

                cin >> startPoint >> endPoint >> hour;

                info.point = endPoint - 1;
                info.hour.insert(hour);

                rInfo.point = startPoint - 1;
                rInfo.hour.insert(hour);

                // 입력의 집 번호는 1-based이므로 C++ vector의 0-based index로 변환
                graph[startPoint - 1].push_back(info);
                rGraph[endPoint - 1].push_back(rInfo);

                if (endPoint == partyPlace)
                { writeIndex[startPoint - 1] = graph[startPoint - 1].size() - 1; }

                if (startPoint == partyPlace)
                { rWriteIndex[endPoint - 1] = rGraph[endPoint - 1].size() - 1; }

            }
            solution(partyPlace - 1, partyPlace - 1, 0, graph, rGraph, writeIndex);
            solution(partyPlace - 1, partyPlace - 1, 0, rGraph, graph, rWriteIndex);

            for (int houseLoop = 0; houseLoop < numOfHouses; houseLoop = houseLoop + 1)
            {

                if(houseLoop == partyPlace - 1)
                { continue; }

                int index = writeIndex[houseLoop];
                int rIndex = rWriteIndex[houseLoop];

                findFastPath.insert(*graph[houseLoop][index].hour.begin() + *rGraph[houseLoop][rIndex].hour.begin());

            }

            answer = *findFastPath.begin();
            std::cout << "#" << test_case << " " << answer << "\n";

            findFastPath.clear();
            rWriteIndex.clear();
            writeIndex.clear();
            rGraph.clear();
            graph.clear();

            rWriteIndex.resize(1000, 0);
            writeIndex.resize(1000, 0);
            rGraph.resize(1000);
            graph.resize(1000);

        }

        // 정상종료시 반드시 0을 리턴해야합니다.
        return 0;

    }

- 코드 해석 : 
    
    사용한 알고리즘은 DFS와 DP 이며, 사용한 자료구조 및 자료형은 vector, set, 구조체이고, 그래프는 2가지로 그렸다.
    
    문제의 그래프와 간선의 방향이 뒤집힌 간선 반전 그래프이다.

    그래프를 2가지로 그린 이유는 다음과 같다.

    간선 반전 그래프의 경우, 간선 반전 그래프를 참고하며 정방향 간선 그래프에 간선을 추가하면, 각 집에서 출발하여 X로 도착하는 노선을 전부 누적할 수 있어서 이다.

    간선이 정방향인 그래프는 DP로 활용할 수 있으며, 간선이 정방향인 그래프를 참고하며, 간선 반전 그래프에 간선을 추가하면, X에서 각 집으로 갈 수 있는 최단 노선을 누적할 수 있기 때문이다. 

    이러한 방법을 기반으로 작성자는 문제가 요구하는 정답 자체를 구하는 코드는 작성할 수 있었으나, 탐색 범위가 너무 넓고 광범위하여 시간 초과로 문제 풀이는 실패 하였다.

##### Padding + Slide Window

    // 작성한 코드
    class Solution 
    {
    public:

        int VoidCheck(char c, int index, bool type)
        {

            int isVoid = 0;
            int notVoid = 0;

            if (type == true)
            { 
                isVoid = index;
                notVoid = 0;
            }
            else
            { 
                isVoid = 0;
                notVoid = index; 
            }

            if (c - ' ' != 0)
            { return notVoid; }

            return isVoid;

        }

        int lengthOfLastWord(string s) 
        {

            // 처음에 문자가 있는 경우
            // 문자열이 3보다 작을 경우 예외처리
            // Padding
            s.insert(0, " ");
            s.insert(0, " ");

            bool findEnd = false;

            int length = s.length();
            int answer = 0;

            int startIndex = length - 1;
            int endIndex = 0;

            int window = 0;
            int slide = 3;

            for(window = length - 1; window >= 0; window = window - slide)
            {


                // 슬라이딩 초과 대비
                if(window < slide - 1)
                { window = 2; }


                // 시작점 찾기
                // OR 논리 연산자의 특이점 사용
                // true일 시 후술 조건들에 대해 실행을 하지 않는 점을 이용하여,
                // 정확한 startIndex를 추출 
                if ( findEnd == false && 
                    ((startIndex = VoidCheck(s[window], window, false)) || 
                    (startIndex = VoidCheck(s[window - 1], window - 1, false)) || 
                    (startIndex = VoidCheck(s[window - 2], window - 2, false))))
                {

                    window = startIndex;
                    findEnd = true;

                }


                // 종료 지점 찾기
                // startIndex와 동일 처리
                // Padding에 의해 맨 첫 글자가 공백이었어도 올바른 값을 출력
                if ( findEnd == true && 
                    ((endIndex = VoidCheck(s[window], window, true)) || 
                    (endIndex = VoidCheck(s[window - 1], window - 1, true)) || 
                    (endIndex = VoidCheck(s[window - 2], window - 2, true))))
                { break; }

            }

            answer = startIndex - endIndex;

            return answer;

        }

    };

- 1번 코드 해석:

    Slide Window와 padding을 이용하여 풀이했다.

    우선 공백 문자를 2번 padding으로 넣어 문자열의 길이가 1혹은 2에 해당하는 상황을 배제하고 크기가 3인 Slide Window를 활용하였다.

    || 연산자의 특이 사항인 선제 조건에서 true가 발생하면 이어지는 조건을 연산하지 않는 방식이용했다.
    
    그렇게 하면 크기 3의 슬라이딩 윈도우에 활용하여 정확한 index를 구할 수 있는데, 이 인덱스의 차를 계산해 길이를 반환했다.

##### String 내부 함수 활용(Parsing)

    class Solution 
    {
    public:

        int lengthOfLastWord(string s) 
        {

            
            string buf;
            buf.assign(s);

            int startS = s.find_last_not_of(' ') + 1;
            s.erase(startS);

            int startBuf = buf.find_last_not_of(' ');
            buf.erase(startBuf);

            int endBuf = buf.find_last_of(' ') + 1;
            buf.erase(endBuf);

            return s.length() - buf.length();

        }

    };
-------------------------------------

- 2번 코드 해석:

    String의 find_last_not_of와 find_last_of를 활용하여
    위치 인덱스를 구하고, s의 복제본을 만들어 해당 부분만 Parsing한 뒤 s의 공백 문자를 삭제하고, 복제본에서 인덱스 위치에서 끝까지 삭제한 뒤 두 문자열의 길이를 뺌으로써 길이를 구하였다.
