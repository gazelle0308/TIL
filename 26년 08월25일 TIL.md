#### N개의 최소 공배수:

  - 해당 문제: https://school.programmers.co.kr/learn/courses/30/lessons/12953

  - 풀이 코드: 
--------------------------------

    // 작성한 코드
    #include <string>
    #include <vector>
    #include <set>

    using namespace std;

    int solution(vector<int> arr) {
        
        int gcd = 0;
        int lcm = 0;
        int answer = 0;
        int remainder = 0;
        
        int buf[2] = {};
        
        set<int> nums;
        
        // 사이즈가 1일 경우의 연산 속도 절약을 위한 예외처리
        if(arr.size() == 1)
        {
            answer = arr[0];
            return answer;
        }
        
        // 초기값 설정(initialization)
        lcm = arr[0];
        
        for(int index = 1; index < arr.size(); index = index + 1)
        {
            
            nums.insert(arr[index]);
            nums.insert(lcm);
            
            
            // 들어가는 2개의 숫자가 같을 경우 
            if(nums.size() == 1)
            { 
                lcm = *nums.begin();
                nums.clear(); 
                continue;
            }

            // lcm 공식을 위한 숫자 버퍼
            buf[0] = *nums.begin();
            buf[1] = *nums.rbegin();
            
            for(int loop = *nums.begin(); loop != 0; loop = remainder)
            {

                remainder = *nums.rbegin() % loop;
                nums.erase(*nums.rbegin());
                nums.insert(remainder);

            }

            gcd = *nums.rbegin();
            lcm = (buf[0] * buf[1])/gcd;

            nums.clear();
            
        }
        
        answer = lcm;
        return answer;
        
    }
------------------------------------------
- 코드 해석 :

    Set의 기반 학습을 목적으로 set의 자동 정렬, 중복 원소 처리, iterator를 통한 원소 접근 및 erase() / insert() 이후의 정렬 상태 유지 등을 직접 활용하였다. 특히 begin()과 rbegin()을 통해 최솟값과 최댓값을 직접 비교하지 않고 얻을 수 있었으며, 이를 유클리드 호제법의 구현에 응용하였다. 

    - 유클리드 호제법:
        
        최대 공약수를 구하는 방법 중 하나이다.
    
        방법은 두 수 a 와 b 가 존재하고, a가 큰 수 b가 작은 수 일 때 a % b를 진행한다.
        
        a를 b로 나눈 나머지를 x라고 하고, a = b, b = x로 갱신하여 x가 0이 될 때까지 반복한다.
        이렇게 반복한 끝에 나머지가 0이 되었을 때의 b가 최대공약수라는 수학적 이론이었다.

        이러한 유클리드 호제법을 이용해 얻은 GCD를 이용하여
        LCM을 얻을 수 있는데 작성자는 해당 방법을 사용하여
        위의 문제를 풀이 하였다.

    - Set의 정렬: 
        
        Set은 추가 설정이 없다면 기본적으로 오름차순 정렬한다. 
        
        작성자는 이를 통해 요소를 2개만 넣음으로써 대소 비교를 할 수 있을 것이라고 생각했으며, 이는 iterator의 응용으로 얻을 수 있었다.
        
        해당 코드에서 사용한 iterator의 응용 방법은 정방향 iterator인 begin()의 *(역참조 연산자)를 통한 작은 값 구하기와 역방향 iterator인 rbegin()과 *(역참조 연산자)를 통한 큰 값 구하기를 활용할 수 있었다.

        또한 해당 코드에서 가장 재미있던 점은 자동 정렬이 가능하니

            remainder = *nums.rbegin() % loop;
            nums.erase(*nums.rbegin());
            nums.insert(remainder);

        이렇게 "위치 기반으로 컨테이너 내부의 구체적인 값 자체를 직접 검색하지 않고, iterator가 가리키는 위치를 통해 필요한 값을 선택할 수 있다" 는 점이었다.

    #include <string>
    #include <vector>
    #include <queue>
    #include <cmath>

    using namespace std;

    int PopLoop(queue<vector<int>>& target)
    {   
        
        int countBuf = target.front().size();
        target.pop();
        return countBuf;
        
    }

    void LogicContinue(vector<int>& answer, int value)
    { answer.push_back(value); }

    vector<int> solution(vector<int> progresses, vector<int> speeds) 
    {
        
        vector<int> answer;
        queue<vector<int>> buffer;
        
        pair<int, int> beforeBuf;
        
        int leftDays;
        
        const int complete = 100;
        
        
        for(int vLoop = 0; vLoop < progresses.size(); vLoop = vLoop + 1)
        {
            
            // 소수점을 요구하는 후속 연산을 위한 double로의 upcast
            double leftProgresses = complete - progresses[vLoop];
            
            // ceil <- cmath의 올림 함수, 소수점 요구
            leftDays = ceil(leftProgresses / speeds[vLoop]);
            
            // 초기값 설정(initialization)
            if(vLoop == 0)
            { 
                beforeBuf = { vLoop, leftDays };
                continue;
            }
            
            if(beforeBuf.second < leftDays)
            { 
                
                buffer.push(vector<int>(progresses.begin() + beforeBuf.first, 
                                        progresses.begin() + vLoop));
                beforeBuf = { vLoop, leftDays }; 
                
            }

        }
        
        buffer.push(vector<int>(progresses.begin() + beforeBuf.first, 
                                progresses.end()));

        
        // 버퍼 비우기, answer 진행 작업 반환
        while(!buffer.empty())
        { LogicContinue(answer, PopLoop(buffer)); }

        return answer;
        
    }
