    /**
    * Definition for singly-linked list.
    * struct ListNode {
    *     int val;
    *     ListNode *next;
    *     ListNode() : val(0), next(nullptr) {}
    *     ListNode(int x) : val(x), next(nullptr) {}
    *     ListNode(int x, ListNode *next) : val(x), next(next) {}
    * };
    */

    class Solution {
    public:
        ListNode* reverseKGroup(ListNode* head, int k) 
        {
            ListNode* answer;

            vector<ListNode*> targetNodes(k);

            ListNode* beforeEnd = nullptr;
            ListNode* bufNext = nullptr;
            ListNode* nowNode = head;

            int endIndex = k - 1;
            int rIndex = 1;

            bool end = false;

            if( k == 1 ){ return head; }

            while(true)
            {

                int index = k - rIndex;

                if(index <= -1) 
                { 

                    rIndex = 1;
                    index = k - rIndex;
                    
                    if(beforeEnd != nullptr) { beforeEnd->next = targetNodes[0]; }
                    beforeEnd = targetNodes[endIndex];

                    if(targetNodes[endIndex] == head)
                    { answer = targetNodes[0]; }

                }

                targetNodes[index] = nowNode;
                if(targetNodes[index]->next == nullptr)
                { break; }
                nowNode = nowNode->next;

                rIndex = rIndex + 1;

                if(index == 0)
                { bufNext = targetNodes[index]->next; }


                int nextIndex = index + 1;
                if(index != endIndex)
                { targetNodes[index]->next = targetNodes[nextIndex]; }

            }

            int index = k - rIndex;

            if(index != 0 && rIndex > 2) // 3개 이상 남을 경우 후 처리
            {
                for(int loop = 1; loop <= rIndex; loop = loop + 1)
                {
                    
                    if(loop == rIndex)
                    { targetNodes[k - loop]->next = nullptr; }

                    if(loop == 1)
                    { bufNext = targetNodes[k - loop]; }
                    else
                    { targetNodes[k - loop + 1]->next = targetNodes[k - loop]; }

                }

                beforeEnd->next = bufNext;

            }
            else if(index == 0) // 정확히 맞아 떨어지는 경우 후 처리
            {
                if(targetNodes[endIndex] == head)
                { answer = targetNodes[0]; }

                for(int loop = 0; loop < endIndex; loop = loop + 1)
                { targetNodes[loop]->next = targetNodes[loop + 1]; }

                targetNodes[endIndex]->next = nullptr;

                if(beforeEnd != nullptr) { beforeEnd->next = targetNodes[0]; }

            }
            else if(index != 0 && rIndex <= 2) // 2개 이하로 남을 경우 후 처리
            { beforeEnd->next = bufNext; }
            return answer;

        }

    };
---------------------------------------------------
- 문제 유형: LinkedList

- 특이 사항: Singly-linked list

- 풀이:
  해당 풀이는 리버스 인덱스와 주어진 k만큼의 유동적인 크기의 vector를 슬라이딩 윈도우로 활용하여, k만큼의 노드에 대해 위치를 실질적으로 반전하고, 다시 Link하는 방식을 활용하고, 반전 중 남은 노드에 대하여 후속처리를 진행 하는 방식으로 풀이 하였다.
