1.알고리즘 학습
-
- 금일 학습한 문제: https://leetcode.com/problems/count-and-say/
- 작성한 코드:
-------------------------------------------------------------------------------
    class Solution 
    {

    public:
        string countAndSay(int n) 
        {
            
            deque<char> slideWindow;
            string answer = "";

            int count = 1;
            
            char bufChar = 0;
    
            for (int loop = 0; loop < n;)
            {

                if(loop == 0)
                { slideWindow.insert(slideWindow.end(), answer.begin(), answer.end()); }

                if(answer.empty() && slideWindow.empty())
                {

                    answer.append(to_string(count));
                    loop = loop + 1;
                    continue;
                    
                }

                if(!slideWindow.empty())
                {

                    if(bufChar != slideWindow.front())
                    { 

                        if(bufChar != 0)
                        {

                            answer.append(to_string(count));
                            answer.push_back(bufChar);

                        }
                        else
                        { answer.clear(); }

                        bufChar = slideWindow.front();
                        count = 1;

                    }
                    else
                    { count = count + 1;}

                    slideWindow.pop_front();

                    if(slideWindow.empty())
                    {
                        answer.append(to_string(count));
                        answer.push_back(bufChar);
                        loop = loop + 1;
                    }

                }
                else
                { 
                    count = 1;
                    bufChar = 0;
                    slideWindow.insert(slideWindow.end(), answer.begin(), answer.end()); 
                }

            }

            return answer;
        }

    };
-----------------------------------------------------------------------------------
- 문제 유형: Run-Length Encoding (RLE)

- deque를 활용하여 문제를 풀어보았으며 문자열을 deque<char>에 복사한 뒤, 앞에서부터 하나씩 pop_front()하며 연속된 문자의 개수를 계산하였다.
