### 1. 알고리즘 스터디

#### 양팔 저울 문제:

  - 해당 문제: https://swexpertacademy.com/main/code/problem/problemDetail.do?contestProbId=AWAe7XSKfUUDFAUw

  - 풀이 코드: 
---------------------------------

    #include <iostream>
    #include <vector>

    using namespace std;


    // weights 
    // first -> 추의 무게
    // second -> 해당 추의 사용 여부

    // dp -> 총 18bit 000000000|000000000 의 2진법 기반 저울 상태 표현
    // Binary -> 10진수 전환 -> 해당 10진수를 인덱스 삼아 저장을 통한 이미지의 데이터화

    // right
    // first -> 우측 무게
    // second -> 비트마스크

    // left
    // first -> 좌측 무게
    // second -> 비트마스크

    // canUsed -> 사용 가능한 추의 갯수

    int DFS(vector<pair<int, bool>>& weights, vector<int>& dp, pair<int, int> right = { 0, 0 }, pair<int, int> left = { 0, 0 }, int canUsed = 9)
    {

        const int factorial[10] = { 1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880 };
        int deleteCase = 0;

        if(dp[(right.second << 9) + left.second] != -1)
        { return dp[(right.second << 9) + left.second]; }

        for (int loop = weights.size() - 1; loop >= 0; loop = loop - 1)
        {

            if (!right.first && !left.first)
            {

                canUsed = weights.size();
                weights[loop].second = true;
                deleteCase = deleteCase + DFS(weights, dp, right, { left.first + weights[loop].first, 0 + (1 << loop) }, canUsed - 1);
                weights[loop].second = false;

            }
            else
            {

                if(weights[loop].second == true)
                { continue; }

                weights[loop].second = true;

                if(dp[(right.second << 9) + (left.second + (1 << loop))] != -1)
                { deleteCase = deleteCase + dp[(right.second << 9) + (left.second + (1 << loop))]; }
                else
                { deleteCase = deleteCase + DFS(weights, dp, right, { left.first + weights[loop].first, left.second + (1 << loop) }, canUsed - 1); }


                if (dp[((right.second + (1 << loop)) << 9) + left.second] != -1)
                {
                    deleteCase = deleteCase + dp[((right.second + (1 << loop)) << 9) + left.second];
                    weights[loop].second = false;
                    continue;
                }

                if (right.first + weights[loop].first > left.first)
                {
                    deleteCase = deleteCase + (1 << (canUsed - 1)) * factorial[canUsed - 1];
                    dp[((right.second + (1 << loop)) << 9) + left.second] = (1 << (canUsed - 1)) * factorial[canUsed - 1];
                }
                else
                {
                    deleteCase = deleteCase + DFS(weights, dp, { right.first + weights[loop].first, right.second + (1 << loop) }, left, canUsed - 1);
                }

                weights[loop].second = false;

            }
            
        }

        dp[(right.second << 9) + left.second] = deleteCase;
        return deleteCase;

    }

    int main(int argc, char** argv)
    {

        const int factorial[10] = { 1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880 };
        int countWeight = 0;

        int T = 0;
        cin >> T;


        for (int loop = 0; loop < T; loop = loop + 1)
        {

            cin >> countWeight;

            vector<pair<int, bool>> weights;
            vector<int> dp(262143, -1);

            // 처음부터 오른 쪽에 올리는 경우의 수 제외 기본 식은 (2 ^ (N - 1)) * N!
            int answer = (1 << (countWeight - 1)) * factorial[countWeight]; 

            for (int index = 0; index < countWeight; index = index + 1)
            {

                int buf = 0;
                cin >> buf;
                weights.push_back({ buf, false });

            }

            int check = DFS(weights, dp);
            answer = answer - check;

            cout << "#" << loop + 1 << " " << answer << "\n";

        }

        return 0;//정상종료시 반드시 0을 리턴해야합니다.

    }

- 코드 해석 : 
    
    DFS + DP + 비트 마스킹를 이용한 문제 풀이로

    가장 기본으로는 우측에 바로 추가 못 올라간다는 점에서 부터 시작하여 기초 수식을
    (2 ^ (N - 1)) * N!로 잡고, 재귀를 통하여 접근하다가 오른 쪽의 무게가 늘어나면
    재귀를 끊어내고 초과할 수 있는 모든 가능성의 수를 빼주는 방식으로 진행하였다.

    또한 추가적으로 재귀를 진행하던 도중 좌측과 우측의 저울의 형상이 기존에 판정했던 것과
    같다면 추가로 파악할 필요가 존재하지 아니하여 해당 형상의 가능성을 가져와 재귀를 진행하지 않고 마무리하는 방식으로 해결하였다.

    해당 형상의 기록은 비트 마스킹을 이용하여 기록하였으며,
    2진법 18비트를 기준으로 표시 후 해당 2진법을 10진법으로 변경하고,
    해당 정수를 인덱스로 삼아 DP로 활용하였다.
